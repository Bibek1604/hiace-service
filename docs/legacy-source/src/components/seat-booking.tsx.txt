'use client';
import { useState } from 'react';
import { useRouter } from 'next/navigation';
import { Armchair, ArrowRight, ShieldCheck, CircleHelp, Wind, Zap, Droplets, MapPin, Clock3 } from 'lucide-react';
import { type TripCard } from '@/lib/catalog';
import { money, nepalTime } from '@/lib/utils';
import { Button } from './ui/button';
import { api } from '@/lib/client';
export function SeatBooking({trip,preview}:{trip:TripCard;preview:boolean}) {
 const [selected,setSelected]=useState<string[]>([]),[error,setError]=useState(''),[busy,setBusy]=useState(false);const router=useRouter();
 async function submit(e:React.FormEvent<HTMLFormElement>){e.preventDefault();setBusy(true);setError('');try{const f=new FormData(e.currentTarget);const b=await api<{id:string;recoveryCode:string;reference:string}>('bookings',{method:'POST',body:JSON.stringify({tripId:trip.id,seats:selected,name:f.get('name'),phone:f.get('phone')})});sessionStorage.setItem(`recovery_${b.id}`,b.recoveryCode);router.push(`/booking/${b.id}`);}catch(e){setError((e as Error).message);setBusy(false);}}
 return <div className="container page-space"><span className="eyebrow">MAKE YOURSELF COMFORTABLE</span><h1 className="page-title">Your journey. Your favorite seat.</h1>{preview&&<p className="notice">Preview mode · Explore the seat layout. Booking requires a connected database.</p>}<div className="booking-layout"><div><article className="card vehicle-detail"><div className="vehicle-cover" style={{backgroundImage:`url(${trip.vehicle.photo})`}}/><div className="padded"><span className="soft-badge">{trip.vehicle.type}</span><h2>{trip.vehicle.name}</h2><p className="muted">{trip.vehicle.operator} · {trip.vehicle.registration}</p><div className="detail-route"><span>{trip.route.origin}</span><ArrowRight/><span>{trip.route.destination}</span></div><p className="muted"><Clock3 size={15}/> {new Date(trip.departure).toLocaleDateString('en-GB',{timeZone:'Asia/Kathmandu',dateStyle:'medium'})} · {nepalTime(trip.departure)} NPT · {Math.floor(trip.route.durationMinutes/60)}h {trip.route.durationMinutes%60}m</p><p className="muted"><MapPin size={15}/> {trip.route.boardingPoint}</p><div className="amenity-chips">{trip.vehicle.amenities.map((a,i)=>{const Icon=[Wind,Zap,Armchair,Droplets][i%4];return <span key={a}><Icon size={14}/>{a}</span>;})}</div>{trip.vehicle.gallery.length>0&&<div className="gallery">{trip.vehicle.gallery.map(url=><img key={url} src={url} alt={`${trip.vehicle.name} interior`} loading="lazy"/>)}</div>}</div></article><div className="card padded seat-panel"><h2>Pick your seat</h2><p className="muted">Tap an available seat. Choose up to 6 seats.</p><div className="seat-legend"><span><i/> Available</span><span><i className="selected"/> Selected</span><span><i className="occupied"/> Unavailable</span></div><div className="seat-vehicle"><div className="driver-seat"><span>FRONT</span><span>◉ Driver</span></div><div className="seat-grid">{trip.vehicle.seats.map((seat,i)=><button key={seat} type="button" aria-label={`Seat ${seat}${trip.occupied.includes(seat)?', unavailable':''}`} aria-pressed={selected.includes(seat)} disabled={trip.occupied.includes(seat)} style={i%3===1?{marginRight:25}:{}} className={`seat ${selected.includes(seat)?'selected':''}`} onClick={()=>setSelected(s=>s.includes(seat)?s.filter(x=>x!==seat):s.length<6?[...s,seat]:s)}><Armchair size={23}/><span>{seat}</span></button>)}</div><small>REAR</small></div><p className="small muted"><CircleHelp size={14}/> Seat availability is checked again when you continue.</p></div></div><form onSubmit={submit} className="card padded booking-summary"><h2>Let’s get you on board.</h2><p className="muted">No account needed. Just a few details.</p><label className="form-label">Passenger full name<input name="name" required minLength={2} maxLength={100} autoComplete="name" placeholder="Your full name"/></label><label className="form-label">Mobile number<div className="phone-input"><span>+977</span><input name="phone" required pattern="9[78][0-9]{8}" maxLength={10} inputMode="tel" autoComplete="tel-national" placeholder="98XXXXXXXX"/></div></label><div className="summary-line"><span>Selected seats</span><strong>{selected.join(', ')||'Choose your seats'}</strong></div><div className="summary-line"><span>Fare per seat</span><strong>{money(trip.fare)}</strong></div><div className="summary-line total"><span>Total fare</span><strong>{money(selected.length*trip.fare)}</strong></div><p className="small muted"><ShieldCheck size={15}/> Your seats are held for 15 minutes while you pay. A receipt upload awaits admin verification.</p>{error&&<p className="error" role="alert">{error}</p>}<div className="sticky-action"><Button disabled={!selected.length||busy||preview} type="submit">{busy?'Holding your seats…':`Continue to payment${selected.length?` · ${money(selected.length*trip.fare)}`:''}`}<ArrowRight size={17}/></Button></div></form></div></div>;
}
