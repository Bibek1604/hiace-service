'use client';
import { useCallback, useEffect, useState } from 'react';
import Link from 'next/link';
import { useRouter } from 'next/navigation';
import { CheckCircle2, Clock3, Download, ShieldCheck, UploadCloud, Copy, ArrowRight } from 'lucide-react';
import { Button } from './ui/button';
import { api } from '@/lib/client';
import { money, nepalTime } from '@/lib/utils';
type Booking={id:string;reference:string;name:string;phone:string;seats:string[];amount:number;status:string;expiresAt:string;trip:{departure:string;route:{origin:string;destination:string;boardingPoint:string};vehicle:{name:string;registration:string}};payments:{id:string;note?:string}[];qr:{id:string;label:string;image:string;accountName:string}[]};
export function Payment({id,statusOnly=false}:{id:string;statusOnly?:boolean}) {
 const[b,setB]=useState<Booking>(),[error,setError]=useState(''),[busy,setBusy]=useState(false),[qr,setQr]=useState(''),[code,setCode]=useState(''),[now,setNow]=useState(Date.now()),[copied,setCopied]=useState(false);const router=useRouter();
 const load=useCallback(async()=>{try{const data=await api<Booking>(`bookings/${id}`);setB(data);setQr(q=>q||data.qr[0]?.id||'');setError('');}catch(e){setError((e as Error).message);}},[id]);
 useEffect(()=>{load();setCode(sessionStorage.getItem(`recovery_${id}`)||'');const timer=setInterval(load,15000);const clock=setInterval(()=>setNow(Date.now()),1000);return()=>{clearInterval(timer);clearInterval(clock);};},[id,load]);
 async function submit(e:React.FormEvent<HTMLFormElement>){e.preventDefault();setBusy(true);setError('');try{const f=new FormData(e.currentTarget);f.set('qrId',qr);await api(`bookings/${id}/payment`,{method:'POST',body:f});router.push(`/booking/${id}/status`);await load();}catch(e){setError((e as Error).message);}finally{setBusy(false);}}
 if(!b)return <div className="container page-space narrow">{error?<><h1>Retrieve your booking</h1><p className="error">{error}</p><Link className="button button-primary" href="/track-booking">Track booking</Link></>:<div className="skeleton" style={{height:400}}/>}</div>;
 const remaining=Math.max(0,new Date(b.expiresAt).getTime()-now);const qrData=b.qr.find(q=>q.id===qr);const state=b.status==='CONFIRMED'?'confirmed':b.status==='REVIEW'?'review':b.status==='HELD'?'held':'expired';
 return <div className="container page-space"><span className="eyebrow">{statusOnly?'YOUR JOURNEY, AT A GLANCE':'ONE LAST STEP'}</span><h1 className="page-title">{statusOnly?'Your booking updates.':'A quick payment. A journey ahead.'}</h1><div className="booking-layout"><div><div className={`card padded status-card ${state}`}><span className="status-icon">{state==='confirmed'?<CheckCircle2 size={30}/>:<Clock3 size={30}/>}</span><div><h2>{state==='confirmed'?'You’re booked. Happy travels!':state==='review'?'Payment received for review':state==='held'?'Your seats are on hold':b.status==='REJECTED'?'Payment could not be verified':'Your seat hold has expired'}</h2><p>{state==='confirmed'?'Your payment was manually verified by our team.':state==='review'?'An admin will check your payment. Your ticket is not confirmed yet.':state==='held'?`Complete payment in ${Math.floor(remaining/60000)}:${String(Math.floor(remaining/1000)%60).padStart(2,'0')}.`:'These seats have been released. If money was deducted, contact the operator with your transaction reference.'}</p></div></div>{code&&<div className="recovery-box"><strong>Keep your private recovery code</strong><p>Use this with your booking reference and phone number to retrieve your booking on another device. Keep it private.</p><code>{code}</code><Button variant="outline" onClick={async()=>{await navigator.clipboard.writeText(`Reference: ${b.reference}\nRecovery code: ${code}`);setCopied(true);}}><Copy size={15}/>{copied?'Copied':'Copy booking details'}</Button></div>}
 {!statusOnly&&state==='held'&&<form className="card padded payment-form" onSubmit={submit}><h2>1. Scan & pay</h2><p className="muted">Use your banking or wallet app to pay <strong>{money(b.amount)}</strong>. Save the QR if you’re using the same phone.</p>{!b.qr.length?<p className="notice">The operator hasn’t added a payment QR yet. Please wait for an available payment account.</p>:<><div className="qr-tabs">{b.qr.map(q=><button type="button" className={`chip ${qr===q.id?'active':''}`} key={q.id} onClick={()=>setQr(q.id)}>{q.label}</button>)}</div>{qrData&&<div className="qr-panel"><img src={`/api/files/${qrData.image}`} alt={`${qrData.label} payment QR`} width={230} height={230}/><strong>{qrData.accountName}</strong><span className="fare">{money(b.amount)}</span><a href={`/api/files/${qrData.image}`} download="payment-qr" className="text-link"><Download size={15}/> Save QR image</a></div>}</>}<h2>2. Share your payment receipt</h2><p className="muted">A screenshot is proof for review. Your booking is confirmed only after our team verifies the actual payment.</p><label className="upload-zone"><UploadCloud size={28}/><strong>Upload payment screenshot</strong><span>PNG, JPEG or WebP · up to 5 MB</span><input type="file" name="screenshot" accept="image/png,image/jpeg,image/webp" required/></label><label className="form-label">Transaction reference <small>(if available)</small><input name="transactionRef" maxLength={100} placeholder="Transaction ID from your payment app"/></label>{error&&<p className="error" role="alert">{error}</p>}<div className="sticky-action"><Button disabled={busy||!qr||!remaining}>{busy?'Uploading receipt…':'Submit for verification'}<ArrowRight size={16}/></Button></div></form>}
 {statusOnly&&state==='held'&&<Link className="button button-primary" href={`/booking/${id}`}>Complete payment <ArrowRight size={16}/></Link>}{['expired'].includes(state)&&<Link className="button button-primary" href="/search">Find another ride</Link>}{state==='review'&&<div className="card padded"><h3>What happens next?</h3><p className="muted">Our team checks the amount and transaction against the operator’s account. Check this page for updates. Review holds expire at departure or after 24 hours, whichever comes first.</p><Button variant="outline" onClick={load}>Refresh status</Button></div>}{state==='confirmed'&&<Button variant="outline" onClick={()=>window.print()}>Print booking confirmation</Button>}</div><aside className="card padded booking-summary"><span className="soft-badge">{b.reference}</span><h2>{b.trip.route.origin} <ArrowRight size={18}/> {b.trip.route.destination}</h2><p>{b.trip.vehicle.name}</p>{[['Passenger',b.name],['Phone',`+977 ${b.phone}`],['Seats',b.seats.join(', ')],['Departure',`${new Date(b.trip.departure).toLocaleDateString('en-GB',{timeZone:'Asia/Kathmandu'})}, ${nepalTime(b.trip.departure)} NPT`],['Boarding',b.trip.route.boardingPoint]].map(([k,v])=><div className="summary-line" key={k}><span>{k}</span><strong>{v}</strong></div>)}<div className="summary-line total"><span>Total fare</span><strong>{money(b.amount)}</strong></div><p className="small muted"><ShieldCheck size={15}/> Private booking. No customer account required.</p>{b.payments[0]?.note&&<p className="notice">Team note: {b.payments[0].note}</p>}</aside></div></div>;
}
