'use client';
import { useState } from 'react';
import Link from 'next/link';
import { ArrowRight, Armchair, Clock3, SlidersHorizontal, Wifi, Wind, Zap } from 'lucide-react';
import { type TripCard } from '@/lib/catalog';
import { money, nepalTime } from '@/lib/utils';
import { Sheet } from './ui/sheet';
export function TripList({ trips }: { trips: TripCard[] }) {
  const [type,setType]=useState('All vehicles'); const [sort,setSort]=useState('departure');
  const filtered=trips.filter(t=>type==='All vehicles'||t.vehicle.type===type).sort((a,b)=>sort==='price'?a.fare-b.fare:a.departure.localeCompare(b.departure));
  const filters=<div className="filter-options">{['All vehicles',...new Set(trips.map(t=>t.vehicle.type))].map(t=><button className={type===t?'chip active':'chip'} key={t} onClick={()=>setType(t)}>{t}</button>)}</div>;
  return <><div className="results-toolbar"><span><strong>{filtered.length}</strong> rides for your journey</span><div><Sheet title="Filter your rides" trigger={<button className="button button-outline"><SlidersHorizontal size={16}/> Filters</button>}>{filters}</Sheet><select aria-label="Sort rides" value={sort} onChange={e=>setSort(e.target.value)}><option value="departure">Earliest departure</option><option value="price">Lowest fare</option></select></div></div><div className="rides-layout"><aside className="filters card"><h3>Make it your kind of ride</h3><p className="muted">Vehicle type</p>{filters}<p className="muted small">All fares include your selected seat. Payment is confirmed after manual verification.</p></aside><div className="ride-list">{!filtered.length&&<div className="card empty"><Armchair/><h2>No rides found</h2><p>Try another date or destination.</p><Link href="/search" className="text-link">See all available rides <ArrowRight size={16}/></Link></div>}{filtered.map(t=><article className="ride-card card" key={t.id}><div className="ride-top"><div className="vehicle-thumb" style={{backgroundImage:`url(${t.vehicle.photo})`}}/><div><span className="soft-badge">{t.vehicle.type}</span><h3>{t.vehicle.name}</h3><p className="muted small">{t.vehicle.operator} · {t.vehicle.registration}</p></div><span className="seats-left"><Armchair size={14}/> {t.available} seats left</span></div><div className="ride-route"><div><strong>{nepalTime(t.departure)}</strong><span>{t.route.origin}</span><small>{t.route.boardingPoint}</small></div><div className="route-line"><small><Clock3 size={12}/> {Math.floor(t.route.durationMinutes/60)}h {t.route.durationMinutes%60}m</small><span>•<i/><ArrowRight size={14}/></span><small>Direct journey</small></div><div><strong>{nepalTime(new Date(new Date(t.departure).getTime()+t.route.durationMinutes*60000))}</strong><span>{t.route.destination}</span><small>{t.route.dropPoint}</small></div></div><div className="ride-bottom"><span className="amenities"><Wind size={16}/><Zap size={16}/><Wifi size={16}/><span>Comfort comes along</span></span><div><span className="fare">{money(t.fare)} <small>/ seat</small></span><Link href={`/trips/${t.id}`} className="button button-primary">Choose seats <ArrowRight size={16}/></Link></div></div></article>)}</div></div></>;
}
