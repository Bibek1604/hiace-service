export async function api<T=Record<string,unknown>>(url:string,options?:RequestInit):Promise<T>{const r=await fetch(`/api/${url}`,{...options,headers:{...(options?.body instanceof FormData?{}:{'Content-Type':'application/json'}),...options?.headers}});const data=await r.json();if(!r.ok)throw new Error(data.error||'Something went wrong.');return data;}
