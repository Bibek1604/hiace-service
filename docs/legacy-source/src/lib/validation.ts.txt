import { z } from 'zod';
export const bookingInput=z.object({tripId:z.string().min(1),name:z.string().trim().min(2).max(100),phone:z.string().regex(/^(?:\+977)?9[78]\d{8}$/, 'Enter a valid Nepal mobile number.'),seats:z.array(z.string().max(8)).min(1).max(6)});
export const vehicleInput=z.object({registration:z.string().min(3).max(40),name:z.string().min(2).max(100),type:z.enum(['Premium Hiace','Tourist Bus','Jeep','Microbus']),photo:z.string().url().refine(s=>s.startsWith('https://')||s.includes('/api/files/')),gallery:z.array(z.string().url()).max(20).default([]),operator:z.string().min(2).max(100),amenities:z.array(z.string().max(80)).max(20),seats:z.array(z.string().regex(/^[A-Z]\d{1,2}$/)).min(1).max(100).refine(a=>new Set(a).size===a.length,'Seat labels must be unique.'),active:z.boolean().default(true)});
export const routeInput=z.object({origin:z.string().min(2).max(80),destination:z.string().min(2).max(80),durationMinutes:z.number().int().min(1).max(3000),distanceKm:z.number().int().min(1).max(3000),boardingPoint:z.string().min(2).max(200),dropPoint:z.string().min(2).max(200)}).refine(a=>a.origin!==a.destination,'Origin and destination must differ.');
export const tripInput=z.object({vehicleId:z.string().min(1),routeId:z.string().min(1),driverId:z.string().optional(),departure:z.iso.datetime(),fare:z.number().int().min(1).max(100000),active:z.boolean().default(true)}).refine(a=>new Date(a.departure)>new Date(),'Departure must be in the future.');
